function eyePosition(state) {
  const s=state || cameraState;
  const r=s.radius,sp=Math.sin(s.phi);
  return [s.target[0]+r*sp*Math.sin(s.theta),s.target[1]+r*Math.cos(s.phi),
    s.target[2]+r*sp*Math.cos(s.theta)];
}
function cameraBasis(state) {
  const s=state || cameraState,eye=eyePosition(s);
  const forward=norm(sub(s.target,eye));
  const right=norm(cross(forward,[0,1,0]));
  const up=norm(cross(right,forward));
  return {eye:eye,forward:forward,right:right,up:up};
}
function project(point,basis,width,height) {
  const delta=sub(point,basis.eye),depth=dot(delta,basis.forward);
  if(depth<=.08) return null;
  const focal=height/(2*Math.tan(42*Math.PI/360));
  return {x:width/2+dot(delta,basis.right)*focal/depth,
    y:height/2-dot(delta,basis.up)*focal/depth,depth:depth};
}
function makeFaces(items,includeNear) {
  const faces=[];
  function face(vertices,normal,item) {
    const center=[0,0,0];
    vertices.forEach(v=>{center[0]+=v[0];center[1]+=v[1];center[2]+=v[2];});
    center[0]/=vertices.length;center[1]/=vertices.length;center[2]/=vertices.length;
    faces.push({vertices:vertices,normal:normal,center:center,item:item});
  }
  items.forEach(item=>{
    if(!includeNear && item.near) return;
    if(item.type==='ring') {
      for(let k=0;k<24;k++) {
        const a=k/24*Math.PI*2,b=(k+1)/24*Math.PI*2;
        face([
          [item.x+Math.cos(a)*item.r,item.y,item.z+Math.sin(a)*item.r],
          [item.x+Math.cos(b)*item.r,item.y,item.z+Math.sin(b)*item.r],
          [item.x+Math.cos(b)*item.inner,item.y,item.z+Math.sin(b)*item.inner],
          [item.x+Math.cos(a)*item.inner,item.y,item.z+Math.sin(a)*item.inner]
        ],[0,1,0],item);
      }
      return;
    }
    const x0=item.x-item.w/2,x1=item.x+item.w/2;
    const y0=item.y-item.h/2,y1=item.y+item.h/2;
    const z0=item.z-item.d/2,z1=item.z+item.d/2;
    face([[x0,y1,z0],[x0,y1,z1],[x1,y1,z1],[x1,y1,z0]],[0,1,0],item);
    face([[x0,y0,z0],[x1,y0,z0],[x1,y0,z1],[x0,y0,z1]],[0,-1,0],item);
    face([[x0,y0,z0],[x0,y1,z0],[x1,y1,z0],[x1,y0,z0]],[0,0,-1],item);
    face([[x1,y0,z1],[x1,y1,z1],[x0,y1,z1],[x0,y0,z1]],[0,0,1],item);
    face([[x0,y0,z1],[x0,y1,z1],[x0,y1,z0],[x0,y0,z0]],[-1,0,0],item);
    face([[x1,y0,z0],[x1,y1,z0],[x1,y1,z1],[x1,y0,z1]],[1,0,0],item);
  });
  return faces;
}

function shade(hex,amount,alpha) {
  const n=parseInt(hex.slice(1),16);
  const r=clamp(Math.round(((n>>16)&255)*amount),0,255);
  const g=clamp(Math.round(((n>>8)&255)*amount),0,255);
  const b=clamp(Math.round((n&255)*amount),0,255);
  return 'rgba('+r+','+g+','+b+','+alpha+')';
}

function renderBasic() {
  if(!ctx || !viewWidth || !viewHeight) return false;
  ctx.setTransform(pixelRatio,0,0,pixelRatio,0,0);
  const bg=ctx.createLinearGradient(0,0,0,viewHeight);
  bg.addColorStop(0,'#f1f1eb');bg.addColorStop(1,'#e2ded3');
  ctx.fillStyle=bg;ctx.fillRect(0,0,viewWidth,viewHeight);
  const basis=cameraBasis();
  const sorted=[];
  basicFaces.forEach(face=>{
    if(dot(face.normal,sub(basis.eye,face.center))<=0) return;
    const points=face.vertices.map(p=>project(p,basis,viewWidth,viewHeight));
    if(points.some(p=>!p)) return;
    if(points.every(p=>p.x<0)||points.every(p=>p.x>viewWidth)||
       points.every(p=>p.y<0)||points.every(p=>p.y>viewHeight)) return;
    sorted.push({face:face,points:points,depth:dot(sub(face.center,basis.eye),basis.forward)});
  });
  sorted.sort((a,b)=>b.depth-a.depth);
  const light=norm([-3,8,5]);
  sorted.forEach(entry=>{
    const f=entry.face,it=f.item;
    const brightness=.69+Math.max(0,dot(f.normal,light))*.34;
    ctx.beginPath();
    entry.points.forEach((p,i)=>{if(i===0) ctx.moveTo(p.x,p.y);else ctx.lineTo(p.x,p.y);});
    ctx.closePath();
    ctx.fillStyle=shade(it.color,brightness,it.opacity);
    ctx.fill();
    if(it.type==='box' && it.tag!=='zone' && it.h>.03) {
      ctx.lineWidth=it.tag==='shear'?1:.45;
      ctx.strokeStyle=it.tag==='shear'?'rgba(112,47,43,.55)':'rgba(62,65,55,.15)';
      ctx.stroke();
    }
  });
  if(showLabels) {
    ctx.textAlign='center';ctx.textBaseline='middle';
    labelPoints().forEach(label=>{
      const p=project(label.p,basis,viewWidth,viewHeight);
      if(!p || p.x<10 || p.x>viewWidth-10 || p.y<12 || p.y>viewHeight-12) return;
      ctx.font=label.color==='#963e38'?'bold 11px sans-serif':'12px sans-serif';
      const width=ctx.measureText(label.name).width+12;
      ctx.fillStyle='rgba(250,248,240,.87)';
      ctx.fillRect(p.x-width/2,p.y-10,width,20);
      ctx.fillStyle=label.color;ctx.fillText(label.name,p.x,p.y);
    });
  }
  ctx.textAlign='left';ctx.font='11px sans-serif';ctx.fillStyle='#697166';
  ctx.fillText('U90 / 实时体块透视 · 非照片级',14,viewHeight-14);
  drawnFaceCount=sorted.length;
  window.__LAB_READY__=true;
  return true;
}

function readEnhancedCamera() {

let loadGeneration=0;
function loadScript(src,validate) {
  return new Promise((resolve,reject)=>{
    const script=document.createElement('script');
    let complete=false;
    const timer=setTimeout(()=>finish(new Error('资源加载超时（6 秒）：'+src)),6000);
    function finish(error) {
      if(complete) return;
      complete=true;clearTimeout(timer);
      script.onload=null;script.onerror=null;
      if(error) {script.remove();reject(error);} else resolve();
    }
    script.src=src;script.async=false;
    script.onload=function() {
      try {
        if(!validate()) throw new Error('资源已响应但接口或版本不符：'+src);
        finish();
      } catch(error) {finish(error);}
    };
    script.onerror=function() {finish(new Error('无法加载同源静态资源：'+src));};
    document.head.appendChild(script);
  });
}
