    stove: Object.freeze({name:"灶台段",w:0.75,h:0.6,counter:true,color:"#f2ede2",stroke:"#8a7a55"}),
    prep: Object.freeze({name:"备餐台面",w:1.0,h:0.6,counter:true,color:"#f2ede2",stroke:"#8a7a55"}),
    trash: Object.freeze({name:"垃圾桶",w:0.35,h:0.35,counter:false,color:"#eff1f1",stroke:"#59666a"})
  });
  const TASK = Object.freeze([
    Object.freeze({at:"fridge",act:"取食材",hands:"full",count:3}),
    Object.freeze({at:"sink",act:"清洗",hands:"free",count:2}),
    Object.freeze({at:"prep",act:"切配",hands:"free",count:2}),
    Object.freeze({at:"stove",act:"烹饪",hands:"free",count:2}),
    Object.freeze({at:"trash",act:"弃料",hands:"full",count:1})
  ]);
  const FIXED = Object.freeze([
    Object.freeze({id:"flue",name:"烟道",x:0.3,y:0.45,w:0.6,h:0.6}),
    Object.freeze({id:"drain",name:"排水立管",x:0.1,y:1.5,w:0.2,h:0.2})
  ]);
  const clone = value => JSON.parse(JSON.stringify(value));
  const round = value => Math.round((value + Number.EPSILON) * 100) / 100;
  const distance = (a,b) => Math.hypot(a.x-b.x,a.y-b.y);
  const typeOf = (id,o) => o.type || id.split("_")[0];
  function box(id,o) {
    const d = DEFINITIONS[typeOf(id,o)];
    const w = o.rot === 90 ? d.h : d.w;
    const h = o.rot === 90 ? d.w : d.h;
    return {id,x:o.x,y:o.y,w,h,l:o.x-w/2,r:o.x+w/2,t:o.y-h/2,b:o.y+h/2};
  }
  function fixedBox(f) {
    return {...f,l:f.x-f.w/2,r:f.x+f.w/2,t:f.y-f.h/2,b:f.y+f.h/2};
  }
  function intersects(a,b) {
    return Math.min(a.r,b.r)-Math.max(a.l,b.l)>EPS &&
           Math.min(a.b,b.b)-Math.max(a.t,b.t)>EPS;
  }
  function outside(b) {
    return b.l < -EPS || b.t < -EPS || b.r > WIDTH+EPS || b.b > DEPTH+EPS;
  }
  function inRange(v,min,max) {return v>=min-EPS && v<=max+EPS;}
  function triangleAccept(legs,total) {
    return {legsOK:legs.every(v=>inRange(v,1.2,2.7)),totalOK:inRange(total,4,6.6)};
  }

  /* 同墙须实际贴齐；先检查 sink→prep→stove 的空间顺序和边缘间隙。
     相互重叠不冒充“零间隙”，副本也不替代这三个任务站。 */
  function continuousChain(scene) {
    const members=["sink","prep","stove"];
    const entries=members.map(id=>({id,o:scene.objects[id],b:box(id,scene.objects[id])}));
    const onWall=(entry,wall)=>{
      const {o,b}=entry;
      if(outside(b))return false;
      if(wall==="north")return o.rot===0 && Math.abs(b.t)<=EPS;
      if(wall==="south")return o.rot===0 && Math.abs(b.b-DEPTH)<=EPS;
      if(wall==="west")return o.rot===90 && Math.abs(b.l)<=EPS;
      return o.rot===90 && Math.abs(b.r-WIDTH)<=EPS;
    };
    for(const wall of ["north","south","west","east"]){
      if(!entries.every(entry=>onWall(entry,wall)))continue;
      const horizontal=wall==="north" || wall==="south";
      const centers=entries.map(({b})=>horizontal?b.x:b.y);
      const sign=Math.sign(centers[1]-centers[0]);
      if(!sign || sign*(centers[2]-centers[1])<=EPS)continue;
      const gaps=[];
      for(let i=1;i<entries.length;i++){
        const a=entries[i-1].b,b=entries[i].b;
        const gap=horizontal
          ? (sign>0?b.l-a.r:a.l-b.r)
          : (sign>0?b.t-a.b:a.t-b.b);
        gaps.push(gap);
      }
      if(gaps.every(gap=>gap>=-EPS && gap<=CHAIN_GAP+EPS)){
        return {connected:true,wall,members:[...members],gapsM:gaps.map(gap=>round(Math.max(0,gap)))};
      }
    }
    return {connected:false,wall:null,members:[...members],gapsM:[]};
  }

  /* 一份计划同时供评价、任务卡、动画与导出消费，绝不按基线名称硬编码次数。
     远距三趟送菜：三次 F→S、两次 S→F；最后一次送达后继续备餐。 */
  function taskPlan(scene) {
    const pickupDistanceM=distance(scene.objects.fridge,scene.objects.sink);
    const oneTrip=pickupDistanceM<=PICKUP_DISTANCE+EPS;
    const chain=continuousChain(scene);
    const steps=oneTrip
      ? [
          {at:"fridge",act:"一次取齐食材",count:3,hands:"full"},
          {at:"sink",act:"清洗",count:2,hands:"free"}
        ]
      : [
          {at:"fridge",act:"取第1批食材",count:1,hands:"full"},
          {at:"sink",act:"送达第1批",count:1,hands:"free"},
          {at:"fridge",act:"返回取第2批",count:1,hands:"full"},
          {at:"sink",act:"送达第2批",count:1,hands:"free"},
          {at:"fridge",act:"返回取第3批",count:1,hands:"full"},
          {at:"sink",act:"送达并清洗",count:2,hands:"free"}
        ];
    steps.push(
      {at:"prep",act:"切配",count:2,hands:"free"},
      {at:"stove",act:"烹饪",count:2,hands:"free"},
      {at:"trash",act:"弃料",count:1,hands:"full"}
    );
    const segments=[];
    for(let i=1;i<steps.length;i++){
      const from=steps[i-1].at,to=steps[i].at;
      const chainStep=chain.connected && chain.members.includes(from) && chain.members.includes(to);
      const handsFull=(from==="fridge" && to==="sink") || (from==="stove" && to==="trash");
      segments.push({
        from,to,
        distanceM:distance(scene.objects[from],scene.objects[to]),
        handsFull,
        chainStep,
        counted:from!==to && !chainStep
      });
    }
    return {
      strategy:oneTrip?"single-pickup":"three-deliveries",
      pickupDistanceM,
      pickupThresholdM:PICKUP_DISTANCE,
      chainGapThresholdM:CHAIN_GAP,
      continuousChain:chain,
      steps,
      segments
    };
  }
  function taskRoute(scene) {return taskPlan(scene).segments;}

  function aisleClearance(scene,boxes) {
    let best = Infinity, measurement = null;
    const obstacles = [...boxes,...FIXED.map(fixedBox)];
    for (const [id,o] of Object.entries(scene.objects)) {
      if (!DEFINITIONS[typeOf(id,o)].counter) continue;
      const a = box(id,o);
      let directions;
      if (o.rot===90) {
        if (Math.abs(a.l)<EPS) directions=["east"];
        else if (Math.abs(a.r-WIDTH)<EPS) directions=["west"];
        else directions=["west","east"];
      } else {
        if (Math.abs(a.t)<EPS) directions=["south"];
        else if (Math.abs(a.b-DEPTH)<EPS) directions=["north"];
        else directions=["north","south"];
      }
      for (const dir of directions) {
        const horizontal = dir==="east" || dir==="west";
        const positive = dir==="east" || dir==="south";
        const start = horizontal ? (positive?a.r:a.l) : (positive?a.b:a.t);
        let gap = Math.max(0, horizontal ? (positive?WIDTH-start:start) : (positive?DEPTH-start:start));
        for (const b of obstacles) {
          if (b.id===id) continue;
          const crossOverlap = horizontal ?
            Math.min(a.b,b.b)-Math.max(a.t,b.t) :
            Math.min(a.r,b.r)-Math.max(a.l,b.l);
          if (crossOverlap<=EPS) continue;
          const low = horizontal?b.l:b.t, high=horizontal?b.r:b.b;
          let d;
          if (positive && high>start+EPS) d=Math.max(0,low-start);
          else if (!positive && low<start-EPS) d=Math.max(0,start-high);
          else continue;
          gap=Math.min(gap,d);
        }
        if (gap<best) {
          best=gap;
          measurement={id,direction:dir,clearanceM:round(gap)};
        }
      }
    }
    return {minClearance:round(Number.isFinite(best)?best:0),measurement};
  }
  function evaluate(scene) {
    const objects=scene.objects;
    const boxes=Object.keys(objects).sort().map(id=>box(id,objects[id]));
    const fixed=FIXED.map(fixedBox), pairs=[];
    for (let i=0;i<boxes.length;i++) {
      for (let j=i+1;j<boxes.length;j++) if (intersects(boxes[i],boxes[j])) pairs.push([boxes[i].id,boxes[j].id]);
      for (const f of fixed) if (intersects(boxes[i],f)) pairs.push([boxes[i].id,f.id]);
    }
    const outIds=boxes.filter(outside).map(b=>b.id);
    const rawLegs=[
      distance(objects.fridge,objects.sink),
      distance(objects.sink,objects.stove),
      distance(objects.stove,objects.fridge)
    ];
    const rawTotal=rawLegs.reduce((a,b)=>a+b,0);
    const legs=rawLegs.map(round),total=round(rawTotal);
    const range=triangleAccept(rawLegs,rawTotal);
    const plan=taskPlan(scene),segments=plan.segments;
    return {
      workTriangle:{legs,total,legRange:[1.2,2.7],totalRange:[4,6.6],...range},
      aisle:aisleClearance(scene,boxes),
      errand:{
        moveCount:segments.filter(segment=>segment.counted).length,
        walkDistanceM:round(segments.reduce((sum,segment)=>sum+segment.distanceM,0)),
        handsFullMoves:segments.filter(segment=>segment.handsFull).length,
        strategy:plan.strategy,
        pickupDistanceM:plan.pickupDistanceM,
        pickupThresholdM:plan.pickupThresholdM,
        continuousChain:plan.continuousChain,
        sequence:plan.steps.map(step=>step.at),
        segments
      },
      collisions:{pairCount:pairs.length,pairs},
      outOfRoom:{count:outIds.length,ids:outIds}
    };
  }
  function baseline(which="A") {
    /* 两版共享精确、无间隙的北墙操作链：
       sink [0.25,1.05] → prep [1.05,2.05] → stove [2.05,2.80]。
       A：西墙中段对侧（东墙中段）；B：北墙最西侧、靠近链头。
       不缩小家具或隐藏固定设施，预置几何相交仍由 evaluate 如实报告。 */
    const positions={
      fridge:which==="B"
        ? {x:0.3,y:0.325,rot:0}
        : {x:2.5,y:1.125,rot:0},
      sink:{x:0.65,y:0.3,rot:0},
      prep:{x:1.55,y:0.3,rot:0},
      stove:{x:2.425,y:0.3,rot:0},
      trash:{x:0.3,y:1.95,rot:0}
    };
    for (const id of Object.keys(positions)) positions[id].type=id;
    return {objects:positions};
  }
  function snap(scene,id,force=false) {
    const result=clone(scene),o=result.objects[id],d=DEFINITIONS[typeOf(id,o)];
    if (!d.counter) return result;
    const walls=[
      {wall:"west",distance:Math.abs(o.x)},
      {wall:"east",distance:Math.abs(WIDTH-o.x)},
      {wall:"north",distance:Math.abs(o.y)},
      {wall:"south",distance:Math.abs(DEPTH-o.y)}
    ].sort((a,b)=>a.distance-b.distance);
    if (!force && walls[0].distance>=0.35-EPS) {
      o.rot=0;
      return outside(box(id,o))?clone(scene):result;
    }
    const wall=walls[0].wall;
    o.rot=(wall==="west" || wall==="east")?90:0;
    const b=box(id,o);
    if (wall==="west" || wall==="east") {
      o.x=wall==="west"?b.w/2:WIDTH-b.w/2;
      o.y=Math.max(b.h/2,Math.min(DEPTH-b.h/2,o.y));
    } else {
      o.y=wall==="north"?b.h/2:DEPTH-b.h/2;
      o.x=Math.max(b.w/2,Math.min(WIDTH-b.w/2,o.x));
    }
    return result;
  }
  function canonicalScene(scene) {
    const objects={};
    Object.keys(scene.objects).sort().forEach(id=>{
      const o=scene.objects[id];
      objects[id]={type:typeOf(id,o),x:o.x,y:o.y,rot:o.rot};
    });
    return {objects};
  }
  function makeProject(scene,versions,savedAt) {
    const c=canonicalScene(scene);
    return {
      schemaVersion:"isd-design-v1",
      courseId:"ISD-COURSE-01",
      labId:"ISD-L04K",
      scene:{unit:"m",origin:"northwest",width:WIDTH,depth:DEPTH},
      objects:Object.entries(c.objects).map(([id,o])=>({id,...o})),
      scenarioVersion:SCENARIO,
      ruleVersion:RULE,
      versions:{A:versions.A?canonicalScene(versions.A):null,B:versions.B?canonicalScene(versions.B):null},
      savedAt
    };
  }
  function validateScene(input) {
    if (!input || typeof input!=="object" || !input.objects || typeof input.objects!=="object" || Array.isArray(input.objects))
      throw new Error("场景对象格式错误。");
    const entries=Object.entries(input.objects);
    if (entries.length<5 || entries.length>60) throw new Error("工作站数量须为5至60个。");
    const objects={};
    for (const [id,o] of entries) {
      if (!/^(fridge|sink|stove|prep|trash)(_[2-9][0-9]*|_1[0-9]+)?$/.test(id)) throw new Error("存在未知工作站编号。");
      if (!o || typeof o!=="object") throw new Error("工作站数据错误。");
      const type=o.type || id.split("_")[0];
      if (!Object.hasOwn(DEFINITIONS,type) || id.split("_")[0]!==type) throw new Error("工作站类型与编号不一致。");
      if (!Number.isFinite(o.x) || !Number.isFinite(o.y) || ![0,90].includes(o.rot)) throw new Error("坐标或旋转角度无效。");
      const n={type,x:o.x,y:o.y,rot:o.rot};
      if (outside(box(id,n))) throw new Error("导入被拒绝：工作站超出房间。");
      objects[id]=n;
    }
    for (const id of Object.keys(DEFINITIONS)) if (!Object.hasOwn(objects,id)) throw new Error("工程缺少任务站："+DEFINITIONS[id].name);
    return canonicalScene({objects});
  }
  function readProject(data) {
    if (!data || data.schemaVersion!=="isd-design-v1" || data.courseId!=="ISD-COURSE-01" || data.labId!=="ISD-L04K")
      throw new Error("不是本实验支持的工程文件。");
    if (data.scenarioVersion!==SCENARIO || data.ruleVersion!==RULE) throw new Error("场景或规则版本不一致，无法保证复算等价。");
    if (!data.scene || data.scene.unit!=="m" || data.scene.width!==WIDTH || data.scene.depth!==DEPTH ||
        (data.scene.origin && data.scene.origin!=="northwest")) throw new Error("房间尺寸、单位或原点不一致。");
    if (!Array.isArray(data.objects)) throw new Error("工作站列表格式错误。");
    const map={};
    for (const o of data.objects) {
      if (!o || typeof o.id!=="string" || Object.hasOwn(map,o.id)) throw new Error("工作站编号缺失或重复。");
      map[o.id]={type:o.type || o.id.split("_")[0],x:o.x,y:o.y,rot:o.rot};
    }
    const scene=validateScene({objects:map});
    const versions={A:null,B:null};
    for (const key of ["A","B"]) if (data.versions?.[key]) versions[key]=validateScene(data.versions[key]);
    return {scene,versions};
  }
  return Object.freeze({
    WIDTH,DEPTH,EPS,SCENARIO,RULE,PICKUP_DISTANCE,CHAIN_GAP,
    DEFINITIONS,TASK,FIXED,clone,round,distance,typeOf,box,intersects,outside,inRange,
    triangleAccept,continuousChain,taskPlan,taskRoute,evaluate,baseline,snap,
    canonicalScene,makeProject,readProject
  });
})();

      $("playStatus").textContent=`${state}：${name(seg.from)} → ${name(seg.to)}${seg.handsFull?" · 双手占用":""}${seg.chainStep?" · 链内步进不计数":""}；已到达${sim.completedSegments}/${sim.segments.length}段，计数${sim.completed}/${report.errand.moveCount}次。`;
    }

    if (drag) return;

    $("pause").disabled=false;$("simulate").textContent="从头模拟";

    schedule();
  }
  function schedule() {
    cancelAnimationFrame(frame);frame=0;
    if (sim?.running && !sim.paused && !document.hidden) frame=requestAnimationFrame(tick);
  }
