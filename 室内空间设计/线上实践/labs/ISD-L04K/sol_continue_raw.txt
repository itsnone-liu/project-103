完成：${sim.completed}次计数移动，共${fmt(sim.walked)}米。`);
          break;
        }
      }
      if (remaining<=E.EPS) break;
    }
    renderPlan();
    if (bounced) renderScores();
    playStatus();
    schedule();
  }

  function stopSimulation() {
    cancelAnimationFrame(frame);
    frame=0;
    lastFrame=null;
    sim=null;
    $("pause").disabled=true;
    $("pause").textContent="暂停";
    $("simulate").textContent="模拟四菜一汤";
  }

  $("simulate").onclick=startSimulation;
  $("pause").onclick=()=>{
    if (!sim?.running) return;
    sim.paused=!sim.paused;
    lastFrame=null;
    $("pause").textContent=sim.paused?"继续":"暂停";
    playStatus();
    schedule();
  };
  $("speed").addEventListener("input",()=>{
    const value=Number($("speed").value);
    if (sim && Number.isFinite(value) && value>0) {
      sim.speed=value;
      lastFrame=null;
    }
  });
  document.addEventListener("visibilitychange",()=>{
    lastFrame=null;
    playStatus();
    schedule();
  });
  window.addEventListener("pagehide",()=>{
    cancelAnimationFrame(frame);
    frame=0;
    lastFrame=null;
  });

  /*
   * 自检使用实际规则层，不另外实现一套“期望结果计算器”。
   * 优先读取规则层公开的预设；若预设仅由界面按钮持有，则通过
   * 相同按钮加载，测试结束后恢复用户场景。
   */
  function runSelfTests() {
    const savedScene=E.clone(scene);
    const savedSelection=selected;
    const results=[];
    let usedPresetControl=false;

    function assert(condition,message) {
      if (!condition) throw new Error(message);
    }
    function test(title,body) {
      try {
        body();
        results.push({title,ok:true});
      } catch (error) {
        results.push({title,ok:false,message:String(error.message||error)});
      }
    }
    function sceneValue(value) {
      return value && typeof value==="object" &&
        value.objects && value.objects.fridge ? E.clone(value) : null;
    }
    function baseline(letter) {
      const keys=[
        letter,letter.toLowerCase(),
        "baseline"+letter,"preset"+letter,"layout"+letter
      ];
      const containers=[
        E.baselines,E.BASELINES,E.presets,E.PRESETS,E.layouts,E.LAYOUTS,E
      ];
      for (const container of containers) {
        if (!container || typeof container!=="object") continue;
        for (const key of keys) {
          let value=container[key];
          if (typeof value==="function") {
            try { value=value.call(container); } catch (_) { continue; }
          }
          const found=sceneValue(value);
          if (found) return found;
        }
      }
      for (const key of [
        "baseline","preset","makeBaseline","makePreset",
        "createBaseline","createPreset"
      ]) {
        if (typeof E[key]!=="function") continue;
        try {
          const found=sceneValue(E[key](letter));
          if (found) return found;
        } catch (_) {
          /* 某些规则层只公开预设数据或界面入口。 */
        }
      }

      const lower=letter.toLowerCase();
      const controls=Array.from(document.querySelectorAll(
        "button,input[type=button],input[type=radio]"
      ));
      const control=controls.find(el=>{
        const id=(el.id||"").replace(/[^a-z0-9]/gi,"").toLowerCase();
        const data=[
          el.dataset.baseline,el.dataset.preset,el.dataset.layout
        ].filter(Boolean).map(value=>String(value).toLowerCase());
        const text=(el.textContent||el.value||"").trim();
        return data.includes(lower) ||
          ["baseline","preset","layout","load"].some(prefix=>id===prefix+lower) ||
          new RegExp("(?:基线|基准|方案|布局|预设)\\s*"+letter+"(?![a-z])","i").test(text) ||
          new RegExp("^"+letter+"\\s*(?:[：:、.（(—-]|方案|布局)","i").test(text);
      });
      assert(control && !control.disabled,"未找到基线"+letter+"的可用预设入口");
      usedPresetControl=true;
      control.click();
      const found=sceneValue(scene);
      assert(found,"基线"+letter+"没有生成有效场景");
      return found;
    }

    stopSimulation();
    try {
      let a=null,b=null;
      test("读取基线 A",()=>{a=baseline("A");});
      test("读取基线 B",()=>{b=baseline("B");});

      function checkBaseline(label,fixture,expectedMoves) {
        assert(fixture,"基线"+label+"不可用");
        const before=JSON.stringify(fixture);
        const evaluation=E.evaluate(fixture);
        const plan=E.taskPlan(fixture);
        assert(
          evaluation.errand.moveCount===expectedMoves,
          `基线${label}：期望${expectedMoves}次，实际${evaluation.errand.moveCount}次`
        );
        assert(
          plan.segments.filter(segment=>segment.counted).length===expectedMoves,
          "任务段的计数标记与基线"+label+"折返数不一致"
        );
        assert(
          plan.segments.every(segment=>!segment.chainStep || !segment.counted),
          "连续链内步进不得计为折返"
        );
        assert(
          plan.segments.every(segment=>
            Number.isFinite(segment.distanceM) && segment.distanceM>=0
          ),
          "任务段距离必须为有限非负数"
        );
        const distance=plan.segments.reduce((sum,segment)=>sum+segment.distanceM,0);
        assert(
          Math.abs(distance-evaluation.errand.walkDistanceM)<=Math.max(E.EPS,1e-6),
          "任务段总距离与评价步行距离不一致"
        );
        assert(
          JSON.stringify(fixture)===before,
          "评价或任务规划不应修改输入场景"
        );
      }

      /* 同墙且相邻间隙不超过 0.05m 的水槽、备菜台、灶台形成连续链。
         链内仍然行走并累加距离，但 counted 必须为 false。 */
      test("基线 A：连续链规则下 moveCount = 6",()=>checkBaseline("A",a,6));
      test("基线 B：连续链规则下 moveCount = 2",()=>checkBaseline("B",b,2));

      test("基线 B：连续链步进保留在动画计划中",()=>{
        assert(b,"基线 B 不可用");
        const steps=E.taskPlan(b).segments.filter(segment=>segment.chainStep);
        assert(steps.length>0,"连续链不能通过删除链内任务段来实现零计数");
        assert(steps.every(segment=>!segment.counted),"链内步进错误计数");
      });

      test("克隆场景可独立编辑",()=>{
        const copy=E.clone(savedScene);
        const originalX=savedScene.objects.fridge.x;
        copy.objects.fridge.x=originalX+.1;
        assert(
          savedScene.objects.fridge.x===originalX,
          "clone 必须隔离对象坐标"
        );
      });
    } finally {
      selected=savedSelection;
      if (usedPresetControl) {
        commit(E.clone(savedScene),"自检结束，已恢复原场景。");
        selected=savedSelection;
      } else {
        scene=E.clone(savedScene);
      }
      report=E.evaluate(scene);
      renderScores();
      renderPlan();
    }

    const failed=results.filter(result=>!result.ok);
    const summary=failed.length
      ? `自检：${results.length-failed.length}/${results.length}项通过。`+
        failed.map(result=>`${result.title}：${result.message}`).join("；")
      : `自检全部通过（${results.length}项）：基线 A 为6次，基线 B 为2次；连续链内步进不计折返。`;
    const output=$("selfTestResults")||$("testResults");
    if (output) output.textContent=summary;
    for (const result of results) {
      if (!result.ok) console.error("[厨房布局自检]",result.title,result.message);
    }
    status(summary);
    return {ok:failed.length===0,results};
  }

  /* 初始化不启动动画；页面隐藏或暂停时也不持续申请动画帧。 */
  (function bootLab() {
    stopSimulation();
    commit(E.clone(scene),"已就绪：选择预设或拖动工作站，再模拟四菜一汤。");

    const testButton=$("selfTest")||$("runTests")||$("runSelfTests");
    if (testButton) testButton.onclick=runSelfTests;

    runSelfTests();
  })();
})();
</script>
